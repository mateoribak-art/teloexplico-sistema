/* Flecha de "aire que sube" (contorno negro + relleno de color). */
const FlechaArriba: React.FC<{ x: number; y: number; alto?: number; color?: string }> = ({ x, y, alto = 260, color = C.rojo }) => {
  const tope = y - alto;
  const cuerpo = `M${x} ${y} v-${alto}`;
  const punta = `M${x - 36} ${tope + 42} L${x} ${tope - 8} L${x + 36} ${tope + 42}`;
  return (
    <g strokeLinecap="round" strokeLinejoin="round" fill="none">
      <path d={cuerpo} stroke={C.tinta} strokeWidth={36} />
      <path d={punta} stroke={C.tinta} strokeWidth={34} />
      <path d={cuerpo} stroke={color} strokeWidth={22} />
      <path d={punta} stroke={color} strokeWidth={20} />
    </g>
  );
};

/* Granizo cayendo: n piedritas en bucle entre yTop e yBot. */
const GranizoCae: React.FC<{ frame: number; n?: number; x0?: number; x1?: number; yTop?: number; yBot?: number; periodo?: number; rmin?: number }> = ({
  frame, n = 7, x0 = 180, x1 = 900, yTop = 980, yBot = 1500, periodo = 40, rmin = 16,
}) => (
  <g>
    {Array.from({ length: n }).map((_, i) => {
      const ciclo = ((frame + i * 11) % periodo) / periodo;
      const x = x0 + (x1 - x0) * ((i + 0.5) / n) + Math.sin(i * 2.3) * 24;
      const y = yTop + (yBot - yTop) * ciclo;
      return <Piedra key={i} cx={x} cy={y} r={rmin + (i % 3) * 7} />;
    })}
  </g>
);
